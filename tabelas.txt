-- 1. Remoção Limpa das Tabelas Existentes (caso já existam no banco)
DROP TABLE IF EXISTS itens CASCADE;
DROP TABLE IF EXISTS subcategoria CASCADE;
DROP TABLE IF EXISTS categoria CASCADE;

-- 2. Criação da Tabela Categoria
CREATE TABLE categoria (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(50) NOT NULL UNIQUE
);

-- 4. Criação da Tabela Subcategoria
CREATE TABLE subcategoria (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(50) NOT NULL UNIQUE,
    categoria_id INTEGER NOT NULL,
    FOREIGN KEY (categoria_id) REFERENCES categoria(id) ON DELETE CASCADE
);

-- 6. Criação da Tabela Itens
CREATE TABLE itens (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    descricao VARCHAR(200),
    preco DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    imagem TEXT NOT NULL,
    z_index INTEGER NOT NULL DEFAULT 1,
    subcategoria_id INTEGER NOT NULL,
    FOREIGN KEY (subcategoria_id) REFERENCES subcategoria(id) ON DELETE CASCADE
);
